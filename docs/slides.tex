\section{技术特色}

\begin{frame}[fragile]
    \frametitle{特色一：Capability-based Handle 能力系统}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item 所有底层资源统一为 \high{对象句柄}
                \item 每个 Handle 附带细粒度 \code{rights} 掩码
                \item 支持 \high{时间到期} 与 \high{操作次数限制}
                \item 集成 \high{Bell-LaPadula} 多级安全标签
                \item 防止 IPC 传递中的能力逃逸
            \end{itemize}
            \vspace{0.3em}
            \small 关键文件：\code{kernel/abi/native/handle\_table.c}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny]
typedef struct a20_handle_entry {

    uint64_t        expiry_tick;

    uint8_t         security_label;

    \end{columns}
\end{frame}

\begin{frame}[fragile]
    \frametitle{特色二：VMO/VMAR 对象化内存模型}
    \begin{itemize}
        \item \high{VMO}（Virtual Memory Object）：物理页容器，可被 resize、共享、按页 fault
        \item \high{VMAR}（Virtual Memory Address Region）：地址空间布局，负责映射与权限
        \item 将物理内存与虚拟地址布局 \high{解耦}，为零拷贝 IPC 提供底座
    \end{itemize}
    \vspace{0.3em}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \small 关键文件：\code{kernel/mm/a20\_vmo.c}、\code{kernel/mm/a20\_vmar.c}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny]

    uint32_t flags;
    a20_handle_t out_handle;
} a20_vm_object_args_t;

        \end{column}

        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]
    \frametitle{特色三：SMP 分级调度器}
    \begin{itemize}
        \item \high{Per-CPU} 独立运行队列，消除全局调度锁
        \item \high{8 级优先级} + bitmap \code{O(1)} 快速选任务
        \item 支持 \high{老化提升}（aging promotion）与 \code{SCHED\_FIFO/RR}
        \item 与 cgroup CPU quota 联动，实现 \high{限流调度}
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/proc/sched.c}、\code{kernel/proc/cg\_cpu.c}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
proc_runq_t rq[NR_CPUS];
uint8_t bitmap[SCHED_LEVELS];
sched_promote_aged_locked(rq);
    \end{lstlisting}
\end{frame}

\begin{frame}
    \frametitle{特色五：事件驱动网络栈}
    \begin{itemize}
        \item \code{lwIP} 以 \code{NO\_SYS=1} 模式集成，避免多线程轮询
        \item \code{kernel/core/progress.c} 提供统一的 \high{bottom-half} 进展机制
        \item virtio-net RX/TX 由 \high{中断 + 事件} 驱动
        \item 调度器切任务前运行 bottom-half，降低上下文切换开销
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/net/lwip\_stack.c}、\code{kernel/core/progress.c}
    \begin{block}{调用路径}
        \footnotesize
        网卡中断 $\rightarrow$ \code{a20\_lwip\_poll()} $\rightarrow$ \code{progress\_run()} $\rightarrow$ socket bottom-half $\rightarrow$ 唤醒等待任务
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{特色九：Native ABI 设计哲学}
    \begin{itemize}
        \item \high{系统调用压缩}：223 个 Linux 系统调用 $\rightarrow$ 109 个 Native syscall（约 2 倍压缩）
        \item \high{统一抽象}：\code{handle\_set\_meta} 替代 chmod/chown/utimes/truncate；\code{handle\_transfer} 统一 splice/sendfile/copy\_file\_range/tee
        \item \high{版本化 ABI}：每个复杂结构以 \code{size + version} 开头，支持运行时协商
        \item 无 fork/exec/signal/ioctl；以 \code{task\_spawn}、Channel、EventQ 替代
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/abi/native/syscall\_table.def}、\code{kernel/include/abi/native/types.h}
\end{frame}

\begin{frame}
    \frametitle{特色十一：安全模型 — rights lattice}
    \begin{itemize}
        \item \high{14 位权利掩码}：READ / WRITE / EXEC / STAT / DUP / TRANSFER / MAP / WAIT / CONNECT / ACCEPT / CONTROL / ADMIN / SIGNAL
        \item \high{权利单调递减}：\code{handle\_dup} 与 IPC 转移只能缩小权利，不能扩大
        \item \high{Bell-LaPadula} 多级安全标签 \{L, M, H\}：no-read-up / no-write-down
        \item \high{Temporal capability}：到期时间与操作次数限制，自动回收或惰性失效
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{docs/native-abi/06-security.md}、\code{kernel/abi/native/handle\_table.c}
\end{frame}

% ==================== 设计文档详解 ====================
\section{设计文档详解}

\begin{frame}
    \frametitle{Native ABI：从 223 到 90 的系统调用压缩}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \footnotesize
            \begin{itemize}
                \item 保留 POSIX 语义核心：\code{open/read/write/close}、\code{mmap/munmap}、\code{clone/wait/exit}
                \item 用对象句柄统一替换 fd 语义：\code{handle\_set\_meta} 替代 chmod/chown/utimes/truncate
                \item 用 \code{handle\_transfer} 统一 splice/sendfile/copy\_file\_range/tee
                \item 取消 fork/exec/signal/ioctl，以 \code{task\_spawn} + Channel + EventQ 替代
            \end{itemize}
        \end{column}
        \begin{column}{0.42\textwidth}
            \footnotesize
            \begin{table}
                \centering
                \renewcommand{\arraystretch}{1.1}
                \begin{tabular}{lr}
                    \toprule
                    \textbf{类别} & \textbf{Native syscall 数} \\
                    \midrule
                    Task \& Process & 12 \\
                    Memory (VMO/VMAR) & 14 \\
                    IPC (Channel/EventQ) & 16 \\
                    IO \& Handle & 22 \\
                    Time \& Sync & 10 \\
                    Misc & 16 \\
                    \midrule
                    \textbf{Total} & \textbf{90} \\
                    \bottomrule
                \end{tabular}
            \end{table}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{docs/native-abi/01-overview.md}、\code{kernel/abi/native/syscall\_table.def}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Handle 生命周期：从创建到回收}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \textbf{创建}：对象构造 $\rightarrow$ 分配 entry $\rightarrow$ 初始化 rights / expiry / label
                \item \textbf{共享传递}：IPC \code{msg\_send} 时源表保留、目标表新增，对象引用计数递增，权利只能缩小
                \item \textbf{复制}：\code{handle\_dup} 可在同进程内复制，rights 子集
                \item \textbf{回收}：关闭引用 $\rightarrow$ entry 置空 $\rightarrow$ 对象引用计数归零后释放
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny]
typedef enum {
    HANDLE_STATE_EMPTY   = 0,
    HANDLE_STATE_ALLOC   = 1,
    HANDLE_STATE_VALID   = 2,
    HANDLE_STATE_EXPIRED = 3
} a20_handle_state_t;
            \end{lstlisting}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \begin{block}{不变式}
        \footnotesize
        任意时刻，一个 Handle 在同一进程 Handle Table 中只存在一份有效引用；IPC 共享传递是原子的，不存在“悬挂引用”。接收方 HT 空间不足时整体返回 \code{NO\_SPACE}，消息留在队列中，杜绝 partial delivery。
    \end{block}
    \small 关键文件：\code{docs/native-abi/03-handles.md}、\code{kernel/abi/native/handle\_table.c}
\end{frame}

\begin{frame}[fragile]
    \frametitle{VMO/VMAR：解耦物理页与地址空间}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \code{a20\_vmo\_create(size, flags, \&h)}：创建物理页容器
                \item \code{a20\_vmar\_map(vmar, h, offset, len, prot, \&addr)}：将 VMO 映射到地址空间
                \item 保护公式：\code{effective\_prot = vmo\_prot $\cap$ vmar\_prot $\cap$ map\_prot}
                \item 共享映射：多个 VMAR 映射同一 VMO，实现零拷贝共享内存
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny]
// VMO resize / commit / decommit
a20_vmo_set_size(h, new_size);
a20_vmo_op_range(h, COMMIT, off, len);
a20_vmo_op_range(h, DECOMMIT, off, len);

// Shared zero-copy IPC
a20_vmo_create(PAGE_SIZE, 0, &vmo);
a20_vmar_map(sender_vmar, vmo, ...);
a20_channel_write(channel, ..., &vmo, 1);
a20_vmar_map(receiver_vmar, vmo, ...);
            \end{lstlisting}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{docs/native-abi/04-memory.md}、\code{kernel/mm/a20\_vmo.c}、\code{kernel/mm/a20\_vmar.c}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Channel/EventQ：异步 IPC 协议}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \textbf{两阶段写入}：reserve 槽位 $\rightarrow$ 拷贝数据 $\rightarrow$ commit；失败可 abort
                \item \textbf{Handle 共享传递}：消息 commit 后源进程仍保留 Handle，对象引用计数递增
                \item \textbf{EventQ}：每个进程一个，Channel 就绪时通过 \code{a20\_event\_signal} 通知
                \item \textbf{等待}：\code{a20\_event\_wait} 多路复用，支持 deadline 和可中断
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny]
// Two-phase channel write
a20_channel_call_args_t args = {
    .channel = ch,
    .data = buf, .data_len = n,
    .handles = &h, .handle_count = 1,
};
a20_channel_call(&args, &reply);
            \end{lstlisting}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \begin{block}{保证}
        \footnotesize
        消息要么完整交付（数据 + 所有 Handle），要么完全不交付；不存在半包或 Handle 重复。
    \end{block}
    \small 关键文件：\code{docs/native-abi/05-ipc.md}、\code{kernel/ipc/a20\_channel.c}
\end{frame}

\begin{frame}
    \frametitle{安全模型：rights lattice 与 MLS}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item 14 位权利掩码构成格，\code{ADMIN} 为最大元，空权利为最小元
                \item \textbf{单调性}：\code{rights(child)} $\subseteq$ \code{rights(parent)}
                \item \textbf{Bell-LaPadula}：低密级主体不可读高密级对象（no-read-up），高密级主体不可写低密级对象（no-write-down）
                \item \textbf{Temporal}：\code{expiry\_tick} 与 \code{remaining\_ops} 提供自动失效
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \footnotesize
            \begin{table}
                \centering
                \renewcommand{\arraystretch}{1.1}
                \begin{tabular}{cl}
                    \toprule
                    \textbf{标签} & \textbf{语义} \\
                    \midrule
                    L (Low) & 普通用户数据 \\
                    M (Medium) & 系统服务中间层 \\
                    H (High) & 内核关键资源 \\
                    \bottomrule
                \end{tabular}
            \end{table}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{docs/native-abi/06-security.md}
\end{frame}

\begin{frame}
    \frametitle{Native 启动协议与 musl 移植}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item 内核通过 \code{a0} 寄存器传递 \code{a20\_start\_info\_t *}，包含初始 Handle 和配置
                \item 初始 Handle：\code{root\_dir}、\code{cwd\_dir}、\code{stdin/stdout/stderr}、\code{self\_task}、\code{event\_queue}
                \item 运行时分层：\code{liba20rt}（syscall 封装）$\rightarrow$ \code{liba20c}（fd 映射、pthread shim）$\rightarrow$ musl
                \item musl 移植：新增 arch 层 \code{a20elf}，替换 syscall 入口，复用 stdio/string/malloc 等 $\sim$45K 行
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \footnotesize
            \begin{block}{目标三元组}
                \code{aarch64-unknown-a20elf}\\
                \code{riscv64-unknown-a20elf}\\
                \code{x86\_64-unknown-a20elf}\\
                \code{loongarch64-unknown-a20elf}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{docs/native-abi/07-startup.md}、\code{user/liba20rt/crt0\_*.S}
\end{frame}

\begin{frame}
    \frametitle{VFS 边界语义与一致性契约}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \code{openat2} 标志：\code{RESOLVE\_NO\_XDEV}、\code{RESOLVE\_BENEATH}、\code{RESOLVE\_NO\_SYMLINKS}
                \item \code{renameat2}：\code{RENAME\_EXCHANGE}、\code{RENAME\_NOREPLACE}、跨 mount 拒绝
                \item 路径解析：处理 mount root、\code{..} 穿越、符号链接深度 \code{MAX\_SYMLINKS=40}
                \item Page Cache：\code{writeback} 与 \code{invalidate} 保证 \code{MAP\_SHARED} 多核一致
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \footnotesize
            \begin{block}{关键不变式}
                \code{lookup} 必须持有父目录引用；\\
                \code{rename} 原子更新目录项与 inode nlink；\\
                跨后端操作返回 \code{-EXDEV}。
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{docs/vfs-edge-semantics.md}、\code{docs/fs-consistency-model.md}
