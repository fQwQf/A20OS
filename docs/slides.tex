\begin{frame}
    \frametitle{分层原则：内部实现 $\otimes$ ABI 薄包装}
    \begin{columns}

                任意新 ABI（含 Linux）直接包装内部机制即可受益，无需复制实现。

\section{并发与调度}

\begin{frame}
    \frametitle{EEVDF 调度器：公平与延迟的统一}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item \high{按权记账}：nice 真实控制 CPU 份额
                \item \high{系统虚拟时间}资格门控 + 虚拟截止时间
                \item RT（级 0，\code{SCHED\_FIFO/RR}）恒优先于 EEVDF
                \item \high{空闲窃取}：本地队列空时拉取远端富余任务
                \item sleeper bonus + \code{EEVDF\_MAX\_LAG} 钳制
                \item \code{/proc/a20/sched\_base\_slice} 运行时切换桌面/HPC
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{EEVDF 核心公式}
                \footnotesize
                \begin{align*}
                    vruntime &\mathrel{+}= dt \times \tfrac{NICE0\_LOAD}{weight} \\
                    vtime &\mathrel{+}= dt \times \tfrac{NICE0\_LOAD}{queued\_weight} \\
                    \text{eligible}:\quad & vruntime \le vtime \\
                    deadline &= vruntime + \tfrac{base\_slice \cdot NICE0\_LOAD}{weight}
                \end{align*}
            \end{block}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/proc/sched.c}；设计：\code{docs/eevdf-scheduler.md}。nice -20/19 权重 43020/88（约 489:1）。
\end{frame}

\begin{frame}
    \frametitle{SMP 负载均衡与持久抢占}
    \begin{itemize}
        \item \high{Per-CPU 运行队列}：消除全局调度锁
        \item 空闲窃取只发生在本地队列空时，\high{只窃 EEVDF 任务、远端 \code{nr\_running} $\geq$ 2、尊重 affinity}
        \item 跨队列迁移按\high{CPU 编号升序}锁定，runqueue 引用全程有效
        \item \high{持久 \code{need\_resched}}：远程唤醒先发布状态，再置标志
        \item IPI 只负责\high{通知}，不在任意中断上下文直接切换；请求在 syscall/trap/timer 返回等安全点消费
        \item 远程唤醒统一走 priority-preempt 快路径，\high{两个 ABI 共享}（pipe/AF\_UNIX/futex/channel）
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/proc/sched.c}、\code{kernel/include/proc/proc.h}；8 核压力门禁 \code{make smoke-sched-stress}
\end{frame}

\section{Native ABI}

\begin{frame}
    \frametitle{Native ABI 设计哲学}
    \begin{itemize}
        \item \high{系统调用压缩}：361 个 Linux syscall $\rightarrow$ 136 个 Native syscall（约 2.7 倍压缩）
        \item \high{统一抽象}：\code{handle\_set\_meta} 替代 chmod/chown/utimes/truncate；\code{handle\_transfer} 统一 splice/sendfile/copy\_file\_range/tee
        \item \high{版本化 ABI}：每个复杂结构以 \code{size + version} 开头，支持运行时协商
        \item 无 fork/exec/signal/ioctl；以 \code{task\_spawn}、Channel、EventQ 替代
        \item syscall 编号稳定分区（0x0000--0x0FFF），演进遵循 E-APPEND / E-DEPRECATE 规则
    \end{itemize}
    \vspace{0.3em}
    \small 关键文件：\code{kernel/abi/native/syscall\_table.def}、\code{kernel/include/abi/native/types.h}
\end{frame}

\begin{frame}
    \frametitle{Native ABI 深化：对象化 Linux 机制}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \high{Pager (0x0D00)}：userfaultfd 的对象化——PAGED VMO + channel 页供给
                \item \high{monitor (0x0D10)}：perf 软件事件的对象化——计数对象可订阅 EventQ
                \item \high{task\_mem\_read/write (0x0211/12)}：process\_vm 的能力化随机访问
                \item \high{vm\_share\_region (0x030A)}：按区间导出 VMO 句柄
                \item \high{EventQ FS/网络事件源}：vnode-keyed FS 事件 + socket/pipe 就绪边沿
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}

                \footnotesize

                    \item \high{应包装}：以更干净对象接口表达
                    \item \high{应拒绝}：SysV/IPC、io\_uring、Landlock 等保持 Linux-only，用 capability 替代
                    \item \high{已有等价物}：pidfd$\rightarrow$TASK handle、memfd$\rightarrow$MEMORY handle

\begin{frame}
    \frametitle{Handle 生命周期与安全模型}
    \begin{columns}
        \begin{column}{0.52\textwidth}
            \footnotesize
            \begin{itemize}
                \item \high{14 位权利掩码}：READ/WRITE/EXEC/STAT/DUP/TRANSFER/MAP/WAIT/CONNECT/ACCEPT/CONTROL/ADMIN/SIGNAL
                \item \high{权利单调递减}：\code{handle\_dup} 与 IPC 转移只能缩小权利
                \item \high{Bell-LaPadula} 多级安全标签 \{L, M, H\}：no-read-up / no-write-down
